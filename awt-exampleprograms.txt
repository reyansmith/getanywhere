Moving Elements by using SetTime out method


<!DOCTYPE html>
<html>
<head>
    <title>Moving Using setTimeout</title>


    <style>
        #box {
            width: 50px;
            height: 50px;
            background: green;
            position: relative;
        }
    </style>
</head>


<body>


    <h3>Move Box Using setTimeout()</h3>


    <div id="box"></div>


    <button onclick="move()">Move Right</button>


    <script>
        let position = 0;


        function move() {


            let box = document.getElementById("box");


            if (position < 300) {


                position = position + 5;
                box.style.left = position + "px";


                setTimeout(move, 50);
            }
        }
    </script>


</body>
</html>










Using Request Animation Method


<!DOCTYPE html>
<html>
<head>
    <title>Moving Using Animation Frame</title>


    <style>
        #box {
            width: 50px;
            height: 50px;
            background: orange;
            position: relative;
        }
    </style>
</head>


<body>


    <h3>Move Box Using requestAnimationFrame()</h3>


    <div id="box"></div>


    <button onclick="startMove()">Move Right</button>


    <script>
        let position = 0;
















        function startMove() {


            let box = document.getElementById("box");
             function animate() {
 
               position = position + 3;


                box.style.left = position + "px";


                if (position < 300) {
                    requestAnimationFrame(animate);
                }
            }


            requestAnimationFrame(animate);
        }
    </script>

Mouse Curser events

            width: 40px;

            border-radius: 50%;

        let box = document.getElementById("box");

Moving elements
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Positioning Elements in JavaScript</title>
    <style>
        #box {
            width: 50px;
            height: 50px;
            background-color: blue;
            position: absolute;
            top: 50px;
            left: 50px;
        }
    </style>
</head>
<body>
    <button onclick="moveBox(20, 20)">Move Right & Down</button>
    <button onclick="moveBox(-20, -20)">Move Left & Up</button>
    <div id="box"></div>


    <script>
        function moveBox(dx, dy) {
            let box = document.getElementById("box");
            let top = parseInt(box.style.top) || 50;
            let left = parseInt(box.style.left) || 50;
            
            box.style.top = (top + dy) + "px";
            box.style.left = (left + dx) + "px";
        }
    </script>
</body>
</html>

        function updateAttendance(statusId) {
            let statusElement = document.getElementById(statusId);


            if (statusElement.textContent === "Absent") {
                statusElement.textContent = "Present";
                statusElement.style.backgroundColor = "lightgreen";
                statusElement.style.color = "black";
                statusElement.style.fontFamily = "Arial, sans-serif";
                statusElement.style.fontSize = "16px";

                statusElement.style.backgroundColor = "lightcoral";
                statusElement.style.color = "white";
                statusElement.style.fontFamily = "Courier New, monospace";
                statusElement.style.fontSize = "14px";
            }
        }
